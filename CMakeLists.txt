cmake_minimum_required(VERSION 3.24)
project(BrowserDataGuard VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The PoC requires x64 Windows. Kernel builds use driver/BdgDriver.vcxproj separately.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_program(BDG_MC mc.exe HINTS "C:/Program Files (x86)/Windows Kits/10/bin/${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}/x64" REQUIRED)
set(BDG_MESSAGES "${CMAKE_CURRENT_BINARY_DIR}/messages")
file(MAKE_DIRECTORY "${BDG_MESSAGES}")
add_custom_command(OUTPUT "${BDG_MESSAGES}/BdgEvents.rc" "${BDG_MESSAGES}/BdgEvents.h" "${BDG_MESSAGES}/MSG00409.bin"
  COMMAND "${BDG_MC}" -h "${BDG_MESSAGES}" -r "${BDG_MESSAGES}" "${CMAKE_CURRENT_SOURCE_DIR}/service/BdgEvents.mc"
  DEPENDS service/BdgEvents.mc VERBATIM)
add_executable(bdg-service service/main.cpp service/identity.cpp service/observe.cpp "${BDG_MESSAGES}/BdgEvents.rc" "${BDG_MESSAGES}/BdgEvents.h")
target_include_directories(bdg-service PRIVATE "${BDG_MESSAGES}")
add_executable(bdg-event-format-tests tests/event_format_tests.cpp "${BDG_MESSAGES}/BdgEvents.rc" "${BDG_MESSAGES}/BdgEvents.h")
target_include_directories(bdg-event-format-tests PRIVATE "${BDG_MESSAGES}")
target_compile_definitions(bdg-event-format-tests PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_link_libraries(bdg-event-format-tests PRIVATE Advapi32)
target_compile_definitions(bdg-service PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_link_libraries(bdg-service PRIVATE FltLib BCrypt Wintrust Crypt32 Advapi32 Shell32 Tdh Ole32)
add_executable(bdg-probe tests/access_probe.cpp)
target_compile_definitions(bdg-probe PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
add_executable(bdg-policy-tests tests/policy_tests.cpp)
add_executable(bdg-user-tests tests/user_tests.cpp service/identity.cpp)
target_compile_definitions(bdg-user-tests PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_link_libraries(bdg-user-tests PRIVATE BCrypt Wintrust Crypt32 Advapi32)
if(MSVC)
  foreach(target bdg-service bdg-probe bdg-policy-tests bdg-user-tests bdg-event-format-tests)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8 /EHsc)
  endforeach()
endif()
enable_testing()
add_test(NAME policy COMMAND bdg-policy-tests)
add_test(NAME windows-event-format COMMAND bdg-event-format-tests)
find_program(BDG_POWERSHELL NAMES pwsh REQUIRED)
add_test(NAME user-identity COMMAND "${BDG_POWERSHELL}" -NoProfile -NonInteractive
  -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/Invoke-UserTests.ps1" -Executable "$<TARGET_FILE:bdg-user-tests>")
add_test(NAME native-probes COMMAND "${BDG_POWERSHELL}" -NoProfile -NonInteractive
  -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/Invoke-ProbeTests.ps1" -ProbeExe "$<TARGET_FILE:bdg-probe>"
  -ScratchRoot "${CMAKE_CURRENT_BINARY_DIR}/probe-tests")
set_tests_properties(native-probes PROPERTIES TIMEOUT 90)
